/-- A 2×2 integer matrix `[[a,b],[c,d]]`. -/
structure Mat where
  a : Int
  b : Int
  c : Int
  d : Int
deriving DecidableEq, Repr

namespace Mat

/-- Matrix product. -/
def mul (X Y : Mat) : Mat :=
  ⟨X.a*Y.a + X.b*Y.c, X.a*Y.b + X.b*Y.d, X.c*Y.a + X.d*Y.c, X.c*Y.b + X.d*Y.d⟩

/-- Matrix sum. -/
def add (X Y : Mat) : Mat := ⟨X.a+Y.a, X.b+Y.b, X.c+Y.c, X.d+Y.d⟩

/-- Identity. -/
def I : Mat := ⟨1, 0, 0, 1⟩

/-- `-I`. -/
def negI : Mat := ⟨-1, 0, 0, -1⟩

end Mat

/-- The complex structure: the generator of continuous (`SO(2)`) reversible 2-level
transport — rotation by 90°. -/
def J : Mat := ⟨0, -1, 1, 0⟩

/-- The sign / reflection carrier: the generator of the **discrete** ℤ/2 real-signed
phase (the only phase the real-bilinear modulus admits, per ForceComplexCarrier §1). -/
def S : Mat := ⟨1, 0, 0, -1⟩

/-- **`J² = -I`** — the rotation generator is a complex structure (continuous phase). -/
theorem J_sq : J.mul J = Mat.negI := by decide

/-- **`S² = +I`** — the sign carrier is an involution of order 2 (discrete ℤ/2). -/
theorem S_sq : S.mul S = Mat.I := by decide

/-- **Continuous vs discrete, concretely.** The continuous phase generator squares to
`-I`; the discrete sign generator to `+I`. Different structures — the algebraic origin of
"the real-signed fiber is discrete `{±1}`, the complex fiber is the continuous circle." -/
theorem continuous_vs_discrete : J.mul J ≠ S.mul S := by decide

/-- The element `a·I + b·J` of the algebra generated by `J`, i.e. `⟨a, -b, b, a⟩`. -/
def cx (a b : Int) : Mat := ⟨a, -b, b, a⟩

theorem cx_one : cx 1 0 = Mat.I := by decide

theorem cx_zero : cx 0 0 = ⟨0, 0, 0, 0⟩ := by decide

theorem cx_add (a b c d : Int) : (cx a b).add (cx c d) = cx (a + c) (b + d) := by
  simp [Mat.add, cx, Mat.mk.injEq]; omega

/-- **The J-algebra multiplies exactly as ℂ.** `(aI+bJ)(cI+dJ) = (ac−bd)I + (ad+bc)J` —
the complex multiplication law, with `J² = -I` doing the work. -/
theorem cx_mul (a b c d : Int) :
    (cx a b).mul (cx c d) = cx (a * c - b * d) (a * d + b * c) := by
  simp only [Mat.mul, cx, Mat.mk.injEq]; refine ⟨?_, ?_, ?_, ?_⟩ <;> ring

theorem cx_injective {a b c d : Int} (h : cx a b = cx c d) : a = c ∧ b = d := by
  have ha : (cx a b).a = (cx c d).a := by rw [h]
  have hc : (cx a b).c = (cx c d).c := by rw [h]
  simp only [cx] at ha hc
  exact ⟨ha, hc⟩

/-- Gaussian-integer multiplication on `ℤ × ℤ` (the integer form of complex
multiplication). -/
def gaussMul (p q : Int × Int) : Int × Int :=
  (p.1 * q.1 - p.2 * q.2, p.1 * q.2 + p.2 * q.1)

/-- **ℂ appears.** `cx` carries the Gaussian-integer product to the matrix product:
`cx p · cx q = cx (p · q)` (with `·` the ℤ[i] product). Together with `cx_add`
(additive), `cx_one` (unital) and `cx_injective` (injective), `cx` is an injective ring
embedding `ℤ[i] ↪ M₂(ℤ)` — the J-generated algebra **is** the (integer) complex numbers,
realized, not posited. -/
theorem cx_realizes_gaussian (p q : Int × Int) :
    (cx p.1 p.2).mul (cx q.1 q.2) = cx (gaussMul p q).1 (gaussMul p q).2 := by
  simp only [gaussMul]; exact cx_mul p.1 p.2 q.1 q.2
