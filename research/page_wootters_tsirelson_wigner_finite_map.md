     $$\Pi_{\mathrm{PW}} = \frac{1}{4}\sum_{k=0}^3 W^k \in M_8(\mathbb{Q}(i)), \qquad \Pi_{\mathrm{PW}}^2 = \Pi_{\mathrm{PW}}, \quad \Pi_{\mathrm{PW}}^\dagger = \Pi_{\mathrm{PW}}, \quad C_{\mathrm{PW}}\Pi_{\mathrm{PW}} = 0, \quad \operatorname{tr}(\Pi_{\mathrm{PW}}) = 2 = \dim(\mathcal{H}_S).$$
