def load() -> SimpleNamespace:
    return SimpleNamespace(**runpy.run_path(str(APP)))


def record() -> dict[str, object]:
    return cast(dict[str, object], json.loads(DATA.read_bytes()))


def rows(data: dict[str, object], key: str) -> list[list[int]]:
    return cast(list[list[int]], data[key])


def complex_vector(values: list[int] | tuple[int, ...]) -> sp.Matrix:
    return sp.Matrix([values[i] + sp.I * values[i+1] for i in range(0, 6, 2)])


def lean_vectors(line: str) -> list[list[int]]:
    return [[int(token) for token in re.findall(r'-?\d+', group)]
            for group in re.findall(r'\(gv ((?:\(?-?\d+\)?\s*){6})\)', line)]


class GaussianExtensionTests(unittest.TestCase):
    def accepted(self, result: Outcome[T]) -> T:
        self.assertIsNone(result.error)
        return cast(T, result.value)

    def refused(self, result: Outcome[T], kind: str | None = None) -> None:
        self.assertIsNotNone(result.error)
        if kind is not None and result.error is not None:
            self.assertEqual(result.error.kind, kind)

    def test_full_symbolic_linear_certificate_and_geometry(self) -> None:
        module = load(); data = record(); rays = rows(data, 'rays')
        matrix = sp.zeros(229, 277); weights = []; targets = []
        for index, (a, b, c, weight) in enumerate(rows(data, 'frames')):
            vectors = [complex_vector(rays[i]) for i in (a, b, c)]
            for i, j in ((0, 1), (0, 2), (1, 2)):
                self.assertEqual(sp.expand((vectors[i].conjugate().T * vectors[j])[0]), 0)
            for i in (a, b, c):
                matrix[index, i] += 1
            weights.append(weight); targets.append(Q(0))
        for index, (ray, sign, weight) in enumerate(rows(data, 'real_values'), 210):
            matrix[index, ray] = 1; weights.append(weight); targets.append(Q(sign))
        witness = sp.Matrix([weights])
        self.assertEqual(witness * matrix, sp.zeros(1, 277))
        self.assertEqual((witness * sp.Matrix(targets))[0], 32)
        self.assertEqual(self.accepted(module.verify(data)),
                         {'rays': 277, 'frames': 210, 'real_values': 19, 'contradiction_rhs': 32})

    def test_all_lean_frame_prescription_and_weight_rows_match(self) -> None:
        text = LEAN.read_text(encoding='utf-8'); data = record(); rays = rows(data, 'rays')
        coefficients = {name: int(value) for value, name in
                        re.findall(r'\(\(?(-?\d+)\)? : ℚ\) \* ([tr]\d+)', text)}
        frames = [line for line in text.splitlines() if re.match(r'  have t\d+ := hF ', line)]
        prescriptions = [line for line in text.splitlines() if re.match(r'    \(r\d+ : f ', line)]
        self.assertEqual((len(frames), len(prescriptions), len(coefficients)), (210, 19, 229))
        for number, (line, frame) in enumerate(zip(frames, rows(data, 'frames'))):
            self.assertEqual(lean_vectors(line), [rays[i] for i in frame[:3]])
            self.assertEqual(coefficients['t'+str(number)], frame[3])
        for number, (line, (index, sign, weight)) in enumerate(zip(prescriptions, rows(data, 'real_values'))):
            self.assertEqual(lean_vectors(line), [rays[index]])
            match = re.search(r'= \(([^:]+) : ℚ\)', line); self.assertIsNotNone(match)
            value = Q(cast(re.Match[str], match).group(1).replace(' ', ''))
            self.assertEqual(3*value-1, sign)
            self.assertEqual(coefficients['r'+str(number)], weight)
        self.assertIn('have contradiction : (0 : ℚ) = 32 / 3', text)

    @PROPERTY
    @given(VECTOR, VECTOR)
    def test_generated_hermitian_inner_matches_symbolic_algebra(self, u: list[int], v: list[int]) -> None:
        expected = sp.expand((complex_vector(u).conjugate().T * complex_vector(v))[0])
        self.assertEqual(load()._inner(tuple(u), tuple(v)), (int(sp.re(expected)), int(sp.im(expected))))

    @PROPERTY
    @given(st.tuples(st.integers(-40, 40), st.integers(-40, 40), st.integers(-40, 40)))
    def test_generated_real_sign_matches_residue_table(self, vector: tuple[int, int, int]) -> None:
        assume(gcd(*vector) == 1)
        odd = [x % 4 for x in vector if x % 2]
        residues: dict[tuple[int, ...], int] = {(1, 1): -1, (3, 3): -1, (1, 3): 1, (3, 1): 1}
        expected = residues.get(tuple(odd), 0)
        ray = tuple(part for value in vector for part in (value, 0))
        self.assertEqual(load()._sign(ray), expected)

    def test_exact_json_numbers_duplicates_unicode_and_syntax(self) -> None:
        for raw in (b'1.0', b'1e0', b'NaN', b'Infinity', b'-Infinity', b'\xff', b'{', b'[] true'):
            self.refused(ExactJsonDocument(raw).parse(), 'input')
        self.refused(ExactJsonDocument(b'{"x":1,"x":2}').parse(), 'duplicate')
        maximum = (1 << 4096)-1
        for value in (0, maximum, -maximum):
            self.assertEqual(self.accepted(ExactJsonDocument(str(value).encode()).parse()), value)
        for encoded in (str(1 << 4096).encode(), b'9'*1235, b'-'+b'9'*1235):
            self.refused(ExactJsonDocument(encoded).parse(), 'resource')
        for invalid in ('{}', bytearray(b'{}'), None):
            self.refused(evaluate(ExactJsonDocument, invalid), 'input')

    def test_document_depth_size_hash_and_ownership(self) -> None:
        self.accepted(ExactJsonDocument(b'['*16+b'0'+b']'*16).parse())
        self.refused(evaluate(ExactJsonDocument, b'['*17+b'0'+b']'*17), 'resource')
        quoted = json.dumps({'text': '"\\'*20 + '['*100 + '}'*100}).encode()
        document = ExactJsonDocument(quoted)
        self.assertEqual(document.sha256, hashlib.sha256(quoted).hexdigest())
        first = cast(dict[str, object], self.accepted(document.parse())); first['text'] = 'changed'
        self.assertEqual(self.accepted(document.parse()), json.loads(quoted))
        with self.assertRaises(FrozenInstanceError):
            document.raw = b'{}'  # type: ignore[misc]
        self.accepted(ExactJsonDocument(b'0'+b' '*(MAX_BYTES-1)).parse())
        self.refused(evaluate(ExactJsonDocument, b' '*(MAX_BYTES+1)), 'resource')

    def test_document_file_lifecycle_and_read_limit(self) -> None:
        self.refused(ExactJsonDocument.read(str(DATA)), 'input')
        with tempfile.TemporaryDirectory() as folder:
            path = Path(folder)/'candidate.json'; self.refused(ExactJsonDocument.read(path), 'resource')
            path.write_bytes(DATA.read_bytes()); document = self.accepted(ExactJsonDocument.read(path))
            path.unlink(); self.assertEqual(document.raw, DATA.read_bytes())
        stream = io.BytesIO(b'0')
        with patch.object(Path, 'open', return_value=stream):
            self.accepted(ExactJsonDocument.read(DATA))
        self.assertTrue(stream.closed)
        with patch.object(Path, 'open', side_effect=PermissionError('refused')):
            self.refused(ExactJsonDocument.read(DATA), 'resource')
        with patch.object(Path, 'open', return_value=io.BytesIO(b'0'*(MAX_BYTES+1))):
            self.refused(ExactJsonDocument.read(DATA), 'resource')

    def test_expected_failure_and_cancellation_propagation(self) -> None:

        self.accepted(module.read_certificate(DATA))
        self.refused(module.read_certificate(None), 'input')
