"""Independent check for the COMP-C7-FIXED-TARGET-OPEN narrowing.

Replays two erasures to the fixed word [0].
Imports checker infrastructure only; no production implementation. These words are not the rule set R0'.
"""

from __future__ import annotations

# Shared checker infrastructure only; no production implementation is imported.
from pathlib import Path as _Path
import sys as _sys
_sys.path.insert(0, str(_Path(__file__).resolve().parents[1]))
from checker_support.boundary import cli, evaluate, require as ensure

def check(cond, msg):
    ensure(cond, msg)

from checker_support.domains import hole


def main():
    failures = 0
    fixed = [0]
    got = hole([1, 0], [], [1], [], [0])
    check(got == fixed, f"erase 1 gave {got}")
    got = hole([2, 0], [], [2], [], [0])
    check(got == fixed, f"erase 2 gave {got}")
    print("checked [1, 0] erases [1] to [0]; [2, 0] erases [2] to [0]; both results are [0]")
    print(f"failures={failures}")
    return failures


if __name__ == "__main__":
    raise SystemExit(cli(main))
